import numpy as np
class WeatherSimulation:
    def __init__(self, transition_probabilities, holding_times):
        self.tp, self.ht = transition_probabilities, holding_times
        for state, probs in self.tp.items():
            if not np.isclose(sum(probs.values()), 1.0):
                raise RuntimeError(f"Probabilities for {state} do not sum to 1.")
        self.current, self.remaining = "sunny", self.ht["sunny"]
    def get_states(self):
        return list(self.tp.keys())
    def current_state(self):
        return self.current
    def next_state(self):
        self.remaining -= 1
        if self.remaining <= 0:
            states, probs = zip(*self.tp[self.current].items())
            self.current = np.random.choice(states, p=probs)
            self.remaining = self.ht[self.current]
        return self.current
    def set_state(self, new_state):
        if new_state not in self.tp:
            raise ValueError("Invalid state")
        self.current, self.remaining = new_state, self.ht[new_state]
    def current_state_remaining_hours(self):
        return self.remaining
    def iterable(self):
        while True:
            yield self.current
            self.next_state()
    def iter_states(self, hours):
        for _ in range(hours):
            yield self.current
            self.next_state()
    def simulate(self, hours):
        counts = {s: 0 for s in self.get_states()}
        for state in self.iter_states(hours):
            counts[state] += 1
        total = sum(counts.values())
        return [counts[s] / total * 100 for s in self.get_states()]
